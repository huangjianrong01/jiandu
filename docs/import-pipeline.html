const ta = document.getElementById('in');
const out = document.getElementById('out');

function render(chapters){
  // 把标记可视化：夹注用淡墨小字，着重用朱色 —— 与 App 里的呈现一致
  const esc = s => s.replace(/[&<>]/g, c => ({'&':'&','<':'<','>':'>'}[c]));
  const fmt = s => esc(s)
    .replace(/\[注\]([\s\S]*?)\[\/注\]/g, '[$1]')
    .replace(/\[高亮\]([\s\S]*?)\[\/高亮\]/g, '$1');
  out.innerHTML = chapters.map(c =>
    `
${esc(c.title)}${
      [...c.content].length} 字
${
      fmt(c.content).replace(/\n/g,'
') || '（空）'}
`
  ).join('') || '（没有解析出内容）';
}

function run(){
  const name = document.querySelector('button[data-sample].on')?.dataset.sample === 'html'
    ? 'demo.html' : (document.querySelector('button[data-sample].on')?.dataset.sample === 'md' ? 'demo.md' : 'demo.txt');
  const bytes = new TextEncoder().encode(ta.value);
  const chapters = JianDuPipeline.parseToJianDuFormatSync(name, bytes);
  render(chapters);
  console.log('管线输出：', chapters);
}

document.querySelectorAll('button[data-sample]').forEach(b => {
  b.onclick = () => {
    document.querySelectorAll('button[data-sample]').forEach(x => x.classList.remove('on'));
    b.classList.add('on');
    ta.value = SAMPLES[b.dataset.sample];
    run();
  };
});
document.getElementById('run').onclick = run;

// 首屏：TXT 样例
document.querySelector('button[data-sample="txt"]').classList.add('on');
ta.value = SAMPLES.txt;
run();
